```python
def get_cdi_data(date_from=None, date_to=None):
    import datetime
    today = datetime.date.today()
    if date_to is None:
        dt_to = today
    else:
        dt_to = datetime.datetime.strptime(date_to, "%Y-%m-%d").date()
    if date_from is None:
        dt_from = today - datetime.timedelta(days=365)
    else:
        dt_from = datetime.datetime.strptime(date_from, "%Y-%m-%d").date()

    fmt = lambda d: d.strftime("%d/%m/%Y")
    url = (
        "https://api.bcb.gov.br/dados/serie/bcdata.sgs.12/dados"
        f"?formato=json&dataInicial={fmt(dt_from)}&dataFinal={fmt(dt_to)}"
    )
    try:
        with urllib.request.urlopen(url, timeout=5) as resp:
            data = json.loads(resp.read().decode())
        accumulated = 1.0
        for item in data:
            accumulated *= 1 + float(item["valor"]) / 100
        return {
            "accumulated": round(accumulated - 1, 6),
            "from": dt_from.isoformat(),
            "to": dt_to.isoformat(),
            "days": len(data),
        }
    except Exception:
        return {
            "accumulated": 0.105,
            "from": dt_from.isoformat(),
            "to": dt_to.isoformat(),
            "days": 252,
            "fallback": True,
        }
```

```python
def get_portfolio_history(tickers, dates):
    if not tickers or not dates:
        return {}
    import datetime
    all_dates = sorted(set(dates))
    min_date = all_dates[0]
    max_dt = datetime.datetime.strptime(max(all_dates), "%Y-%m-%d") + datetime.timedelta(days=7)
    max_date = max_dt.strftime("%Y-%m-%d")
    result = {}
    for ticker in tickers:
        yf_ticker = ticker if ("." in ticker or "=" in ticker) else ticker + ".SA"
        try:
            hist = yf.download(yf_ticker, start=min_date, end=max_date, progress=False, auto_adjust=True)
            if hist.empty:
                hist = yf.download(ticker, start=min_date, end=max_date, progress=False, auto_adjust=True)
            ticker_prices = {}
            for date_str in all_dates:
                date_dt = datetime.datetime.strptime(date_str, "%Y-%m-%d")
                subset = hist[hist.index >= date_dt]
                ticker_prices[date_str] = float(subset["Close"].iloc[0]) if not subset.empty else None
            result[ticker] = ticker_prices
        except Exception:
            result[ticker] = {d: None for d in all_dates}
    return result
```

```python
def do_POST(self):
    length = int(self.headers.get("Content-Length", 0))
    body = self.rfile.read(length)
    try:
        payload = json.loads(body)
    except Exception:
        self._send_json({"code": "BAD_REQUEST"}, status=400)
        return
    path = self.path.split("?")[0].rstrip("/")
    if path == "/api/portfolio/history":
        tickers = payload.get("tickers", [])
        dates   = payload.get("dates", [])
```

```javascript
// src/portfolio-store.js
const KEYS = {
  operations: 'portfolio_operations',
  fixedIncome: 'portfolio_fixed_income',
}

function _load(key) {
  try { return JSON.parse(localStorage.getItem(key) || '[]') }
  catch { return [] }
}
function _save(key, data) {
  localStorage.setItem(key, JSON.stringify(data))
}

export function getOperations() { return _load(KEYS.operations) }

export function saveOperation(op) {
  const ops = getOperations()
  const idx = ops.findIndex(o => o.id === op.id)
  if (idx >= 0) ops[idx] = op; else ops.push(op)
  _save(KEYS.operations, ops)
}

export function deleteOperation(id) {
  _save(KEYS.operations, getOperations().filter(o => o.id !== id))
}

export function getFixedIncome() { return _load(KEYS.fixedIncome) }

export function saveFixedIncomeTitle(title) {
  const titles = getFixedIncome()
  const idx = titles.findIndex(t => t.id === title.id)
  if (idx >= 0) titles[idx] = title; else titles.push(title)
  _save(KEYS.fixedIncome, titles)
}

export function deleteFixedIncomeTitle(id) {
  _save(KEYS.fixedIncome, getFixedIncome().filter(t => t.id !== id))
}

export function addDeposit(titleId, deposit) {
  const titles = getFixedIncome()
  const title = titles.find(t => t.id === titleId)
  if (!title) return
  title.deposits.push(deposit)
  _save(KEYS.fixedIncome, titles)
}

export function deleteDeposit(titleId, depositId) {
  const titles = getFixedIncome()
  const title = titles.find(t => t.id === titleId)
  if (!title) return
  title.deposits = title.deposits.filter(d => d.id !== depositId)
  _save(KEYS.fixedIncome, titles)
}

export function generateId() {
  return crypto?.randomUUID?.() ?? Math.random().toString(36).slice(2) + Date.now().toString(36)
}
```

```javascript
// src/portfolio-engine.js

/** Returns average buy price for a ticker's operation history. */
export function calcPrecoMedio(operations) {
  const buys = operations.filter(o => o.type === 'buy')
  if (!buys.length) return null
  const totalCost = buys.reduce((sum, o) => sum + o.qty * o.price, 0)
  const totalQty  = buys.reduce((sum, o) => sum + o.qty, 0)
  return totalCost / totalQty
}

/** Returns simple return ratio given average cost and current price. */
export function calcRetornoSimples(precoMedio, cotacaoAtual) {
  if (precoMedio == null || cotacaoAtual == null) return null
  return (cotacaoAtual - precoMedio) / precoMedio
}

/**
 * Returns net holdings map { ticker: netQty } for positions still open.
 * Sells reduce qty; fully-sold tickers are excluded.
 */
export function calcHoldings(operations) {
  const h = {}
  for (const op of operations) {
    h[op.ticker] = (h[op.ticker] ?? 0) + (op.type === 'buy' ? op.qty : -op.qty)
  }
  return Object.fromEntries(Object.entries(h).filter(([, q]) => q > 0))
}
```

```javascript
// Append to src/portfolio-engine.js

/**
 * Time-Weighted Rate of Return.
 * @param {Array<{startValue:number, endValue:number}>} subPeriods
 */
export function calcTWRR(subPeriods) {
  if (!subPeriods || !subPeriods.length) return null
  let product = 1
  for (const { startValue, endValue } of subPeriods) {
    if (startValue <= 0) continue
    product *= 1 + (endValue - startValue) / startValue
  }
  return product - 1
}

/**
 * Build sub-periods from operations and historical prices.
 * @param {Array}  operations      - all ops, any order
 * @param {Object} historicalPrices - { ticker: { "YYYY-MM-DD": price } }
 * @param {Object} currentPrices   - { ticker: currentPrice }
 */
export function buildTWRRSubPeriods(operations, historicalPrices, currentPrices) {
  if (!operations.length) return []
  const sorted = [...operations].sort((a, b) => a.date.localeCompare(b.date))
  const uniqueDates = [...new Set(sorted.map(o => o.date))]

  const portfolioValue = (ops, date, useCurrentForFinalDate) => {
    const holdings = calcHoldings(ops)
    return Object.entries(holdings).reduce((sum, [ticker, qty]) => {
      const price = useCurrentForFinalDate
        ? (currentPrices[ticker] ?? 0)
        : (historicalPrices[ticker]?.[date] ?? currentPrices[ticker] ?? 0)
      return sum + qty * price
    }, 0)
  }

  const subPeriods = []
  for (let i = 0; i < uniqueDates.length; i++) {
    const startDate = uniqueDates[i]
    const isLast    = i === uniqueDates.length - 1
    const opsUpToStart = sorted.filter(o => o.date <= startDate)
    const opsUpToEnd   = isLast ? sorted : sorted.filter(o => o.date < uniqueDates[i + 1])
    const startValue = portfolioValue(opsUpToStart, startDate, false)
    const endValue   = portfolioValue(opsUpToEnd, uniqueDates[i + 1], isLast)
    if (startValue > 0) subPeriods.push({ startValue, endValue })
  }
  return subPeriods
}
```

```javascript
// Append to src/portfolio-engine.js

/**
 * Project current value of a single deposit.
 * @param {object} p
 * @param {number} p.amount
 * @param {'cdi_pct'|'ipca_plus'|'prefixado'|'manual'} p.rateType
 * @param {number} p.baseRate       - 110 = 110% CDI; 6.2 = IPCA+6.2%; 12.5 = 12.5% prefixado
 * @param {number} p.cdiAccumulated - 0–1 (e.g., 0.105 for 10.5% CDI in period)
 * @param {number} p.ipcaAccumulated - 0–1 (for ipca_plus only)
 * @param {number} p.daysElapsed
 * @param {number} [p.manualCurrentValue]
 */
export function projectDeposit(p) {
  const { amount, rateType, baseRate, cdiAccumulated = 0, ipcaAccumulated = 0,
          daysElapsed = 0, manualCurrentValue } = p
  if (rateType === 'manual') return manualCurrentValue ?? amount
  if (daysElapsed <= 0) return amount
  const yearFraction = daysElapsed / 365
  switch (rateType) {
    case 'cdi_pct':
      return amount * (1 + (baseRate / 100) * cdiAccumulated)
    case 'prefixado':
      return amount * Math.pow(1 + baseRate / 100, yearFraction)
    case 'ipca_plus': {
      const totalRate = (1 + ipcaAccumulated) * (1 + baseRate / 100 * yearFraction) - 1
      return amount * (1 + totalRate)
    }
    default:
      return amount
  }
}

/**
 * Aggregate all deposits of a title into totals.
 * @param {object} title
 * @param {object} ratesForPeriod  - { cdiAccumulated, ipcaAccumulated }
 * @param {string} today           - "YYYY-MM-DD"
 */
export function aggregateTitle(title, ratesForPeriod, today) {
  let totalInvested  = 0
  let totalProjected = 0
  for (const dep of title.deposits) {
    const depositDate  = dep.date
    const daysElapsed  = Math.max(0, Math.floor((new Date(today) - new Date(depositDate)) / 86400000))
    const rateType     = title.rateType
    const baseRate     = dep.rateOverride ?? title.baseRate
    totalInvested  += dep.amount
    totalProjected += projectDeposit({
      amount: dep.amount, rateType, baseRate,
      cdiAccumulated:  ratesForPeriod.cdiAccumulated  ?? 0,
      ipcaAccumulated: ratesForPeriod.ipcaAccumulated ?? 0,
      daysElapsed,
      manualCurrentValue: dep.manualCurrentValue,
    })
  }
  return {
    totalInvested,
    totalProjected,
    retorno: totalInvested > 0 ? (totalProjected - totalInvested) / totalInvested : null,
  }
}
```

```javascript
// src/b3-csv-parser.js
const KNOWN_ETFS = new Set([
  'BOVA11','SMAL11','IVVB11','HASH11','DIVO11','FIND11','MATB11','ECOO11',
  'BBSD11','BBVO11','SPXI11','FIXA11','GOLD11','XFIX11',
])

export function inferAssetClass(ticker) {
  if (KNOWN_ETFS.has(ticker)) return 'etf'
  if (/11$/.test(ticker)) return 'fii'
  return 'acao_br'
}

function parseBRNumber(s) {
  if (!s) return 0
  // "3.420,00" → 3420.00  |  "34,20" → 34.20
  return parseFloat(s.replace(/\./g, '').replace(',', '.')) || 0
}

function parseBRDate(s) {
  // "15/01/2024" → "2024-01-15"
  const [d, m, y] = s.trim().split('/')
  return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
}

export function parseB3CSV(csvText) {
  const lines = csvText.trim().split(/\r?\n/)
  const headerIdx = lines.findIndex(
    l => l.includes('Data') && (l.includes('Quantidade') || l.includes('Negóci'))
  )
  if (headerIdx < 0) throw new Error('Formato B3 não reconhecido — cabeçalho não encontrado')

  const sep = lines[headerIdx].includes(';') ? ';' : ','
  const headers = lines[headerIdx].split(sep).map(h => h.trim().replace(/"/g, ''))

  const col = name => headers.findIndex(h => h.toLowerCase().includes(name.toLowerCase()))
  const idxDate   = col('Data')
  const idxType   = col('Movimenta')
  const idxTicker = col('Código') >= 0 ? col('Código') : col('Codigo')
  const idxQty    = col('Quantidade')
  const idxPrice  = col('Preço') >= 0  ? col('Preço')  : col('Preco')

  if (idxDate < 0 || idxTicker < 0 || idxQty < 0) {
    throw new Error('Colunas obrigatórias ausentes: Data, Código de Negociação, Quantidade')
  }

  const ops = []
  for (let i = headerIdx + 1; i < lines.length; i++) {
    const row = lines[i].split(sep).map(c => c.trim().replace(/"/g, ''))
    if (row.length < 3) continue
    const ticker = row[idxTicker]?.toUpperCase()
    if (!ticker) continue
    const qty   = parseBRNumber(row[idxQty])
    const price = idxPrice >= 0 ? parseBRNumber(row[idxPrice]) : 0
    if (qty <= 0) continue
    const rawType = idxType >= 0 ? row[idxType] : ''
    ops.push({
      id:         null,
      date:       parseBRDate(row[idxDate]),
      ticker,
      assetClass: inferAssetClass(ticker),
      type:       rawType.toLowerCase().includes('venda') ? 'sell' : 'buy',
      qty,
      price,
      currency:   'BRL',
      fees:       0,
    })
  }
  return ops
}
```

```javascript
import { calcPrecoMedio, calcHoldings } from './src/portfolio-engine.js'

let _assetFilter = 'all'
const _quoteCache = {}

async function fetchQuote(ticker) {
  if (_quoteCache[ticker]) return _quoteCache[ticker]
  try {
    const r = await fetch(`/api/quote/${ticker}`)
    const d = await r.json()
    if (d.price) { _quoteCache[ticker] = d; return d }
  } catch {}
  return null
}

function dcfStatus(precoMedio, fairPrice) {
  if (!fairPrice) return '<span style="color:var(--text-muted);font-size:11px">sem DCF</span>'
  if (precoMedio > fairPrice)              return '<span class="status status-caro">● Caro</span>'
  if (precoMedio < fairPrice * 0.9)        return '<span class="status status-barato">● Barato</span>'
  return '<span class="status status-justo">● Justo</span>'
}

async function renderAtivos() {
  const ops = getOperations()
  const holdings = calcHoldings(ops)
  const watchlist = JSON.parse(localStorage.getItem('dcf_watchlist') || '{}')
  const tbody = document.getElementById('asset-tbody')
  const empty = document.getElementById('asset-empty')

  const tickers = Object.keys(holdings).filter(
    t => _assetFilter === 'all' ||
         ops.find(o => o.ticker === t)?.assetClass === _assetFilter
  )
  if (!tickers.length) { tbody.innerHTML = ''; empty.style.display = 'block'; return }
  empty.style.display = 'none'

  // Skeleton while loading
  tbody.innerHTML = tickers.map(t => `<tr id="row-${t}">
    <td><strong>${t}</strong></td>
    <td colspan="8"><span class="skel" style="display:inline-block;width:80%;height:14px"></span></td>
  </tr>`).join('')

  for (const ticker of tickers) {
    const tickerOps   = ops.filter(o => o.ticker === ticker)
    const assetClass  = tickerOps[tickerOps.length - 1]?.assetClass ?? 'acao_br'
    const qty         = holdings[ticker]
    const precoMedio  = calcPrecoMedio(tickerOps)
    const quote       = await fetchQuote(ticker)
    const cotacao     = quote?.price ?? null
    const valorCarteira = cotacao != null ? qty * cotacao : null
    const retorno     = precoMedio && cotacao
      ? (cotacao - precoMedio) / precoMedio : null
    const wlEntry   = watchlist[ticker]
    const fairPrice = wlEntry?.fairPrice ?? null

    const retStr = retorno != null
      ? `<span style="color:${retorno >= 0 ? 'var(--green)' : 'var(--red)'}">
           ${retorno >= 0 ? '+' : ''}${(retorno * 100).toFixed(1)}%</span>`
      : '—'
    const fairStr = fairPrice
      ? fBRL(fairPrice)
      : `<a href="index.html?ticker=${ticker}" style="color:var(--text-muted);font-size:11px">→ calcular</a>`

    document.getElementById(`row-${ticker}`).innerHTML = `
      <td><strong>${ticker}</strong></td>
      <td><span class="badge ${CLASS_BADGE[assetClass]}">${CLASS_LABELS[assetClass]}</span></td>
      <td class="mono right">${qty}</td>
      <td class="mono right">${precoMedio != null ? fBRL(precoMedio) : '—'}</td>
      <td class="mono right">${cotacao != null ? fBRL(cotacao) : '<span class="skel" style="display:inline-block;width:60px;height:13px"></span>'}</td>
      <td class="mono right">${valorCarteira != null ? fBRL(valorCarteira) : '—'}</td>
      <td class="mono right">${retStr}</td>
      <td class="mono right">${fairStr}</td>
      <td>${dcfStatus(cotacao, fairPrice)}</td>`
  }
}
```

```javascript
import { parseB3CSV } from './src/b3-csv-parser.js'

let _csvParsedOps = []

function dedupKey(op) {
  return `${op.date}|${op.ticker}|${op.type}|${op.qty}|${op.price}`
}

document.getElementById('btn-import-csv').onclick = () => {
  document.getElementById('csv-file-input').click()
}

document.getElementById('csv-file-input').addEventListener('change', async e => {
  const file = e.target.files[0]
  if (!file) return
  const text = await file.text()
  try {
    const parsed = parseB3CSV(text)
    const existing = new Set(getOperations().map(dedupKey))
    const newOps = parsed.filter(o => !existing.has(dedupKey(o)))

    _csvParsedOps = parsed
    const dup = parsed.length - newOps.length

    document.getElementById('csv-summary').textContent =
      `${parsed.length} operações encontradas — ${newOps.length} novas, ${dup} já existentes (pré-desmarcadas).`

    const tbody = document.getElementById('csv-preview-tbody')
    tbody.innerHTML = parsed.map((op, i) => {
      const isDup = existing.has(dedupKey(op))
      return `<tr>
        <td><input type="checkbox" data-csv-idx="${i}" ${isDup ? '' : 'checked'}></td>
        <td class="mono">${op.date}</td>
        <td><strong>${op.ticker}</strong></td>
        <td><span class="badge ${op.type === 'buy' ? 'badge-compra' : 'badge-venda'}">${op.type === 'buy' ? 'C' : 'V'}</span></td>
        <td class="mono right">${op.qty}</td>
        <td class="mono right">${fBRL(op.price)}</td>
      </tr>`
    }).join('')

    document.getElementById('modal-csv').style.display = 'flex'
  } catch (err) {
    alert('Erro ao ler o arquivo: ' + err.message)
  }
  e.target.value = ''  // reset input so same file can be re-imported
})

document.getElementById('csv-check-all').addEventListener('change', e => {
  document.querySelectorAll('[data-csv-idx]').forEach(cb => { cb.checked = e.target.checked })

document.getElementById('btn-cancel-csv').onclick = () => {
  document.getElementById('modal-csv').style.display = 'none'
}

document.getElementById('btn-confirm-csv').onclick = () => {
  const checked = [...document.querySelectorAll('[data-csv-idx]:checked')]
    .map(cb => parseInt(cb.dataset.csvIdx))
  const existing = new Set(getOperations().map(dedupKey))
  let imported = 0
  for (const idx of checked) {
    const op = { ..._csvParsedOps[idx], id: generateId() }
```
